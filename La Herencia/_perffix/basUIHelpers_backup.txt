' Convierte Double a string SQL con punto decimal
Private Function SqlNum(ByVal d As Variant) As String
    If IsNull(d) Then
        SqlNum = "Null"
    Else
        SqlNum = Replace(CStr(CDbl(d)), ",", ".")
    End If
End Function

' Convierte texto a literal SQL (con comillas escapadas) o Null si vacío
Private Function sqlTxt(ByVal s As Variant) As String
    Dim t As String
    t = Trim$(Nz(s, vbNullString))
    If t = vbNullString Then
        sqlTxt = "Null"
    Else
        sqlTxt = "'" & Replace(t, "'", "''") & "'"
    End If
End Function

' ¿Existe un control con ese nombre?

    Dim c As Control: Set c = frm.Controls(ctlName)
    ControlExists = (Err.Number = 0)

End Function

' Devuelve un control por nombre (o Nothing si no existe)
Public Function TryGetControl(frm As Access.Form, ctlName As String) As Access.Control
    On Error Resume Next
    Set TryGetControl = frm.Controls(ctlName)
    Err.Clear
End Function

' Busca el subform embebido por nombre de CONTROL o por SourceObject (Form.XYZ)
Public Function TryGetSubform(frm As Access.Form, subName As String) As Control
    Dim ctl As Control, so As String
    If frm Is Nothing Then Exit Function
    If LenB(Nz(subName, "")) = 0 Then Exit Function

    On Error Resume Next
    Set TryGetSubform = frm.Controls(subName)
    If Not TryGetSubform Is Nothing Then
        Err.Clear
        Exit Function
    End If
    Err.Clear

    For Each ctl In frm.Controls
        If ctl.ControlType = acSubform Then
            so = ctl.SourceObject
            If LCase$(Left$(so, 5)) = "form." Then so = Mid$(so, 6)
            If StrComp(so, subName, vbTextCompare) = 0 Then
                Set TryGetSubform = ctl
                Exit Function
            End If
        End If
    Next ctl
End Function

' Requery seguro de un subform (no rompe si no existe)
Public Sub SafeRequerySub(frm As Object, subCtlName As String)
    Dim s As Control
    Set s = TryGetSubform(frm, subCtlName)
    If Not s Is Nothing Then
        On Error Resume Next
        s.Requery
        s.Form.Requery
        Err.Clear
    End If
End Sub

' Devuelve el nombre del próximo control editable según TabIndex
Public Function NextEditableControlName(frm As Form, cur As Control) As String
    Dim idx As Long: idx = cur.TabIndex
    Dim nextIdx As Long: nextIdx = 99999
    Dim firstIdx As Long: firstIdx = 99999
    Dim nextName As String: nextName = vbNullString
    Dim firstName As String: firstName = vbNullString
    Dim ctl As Control

    For Each ctl In frm.Controls
        If ctl.TabStop And ctl.Visible And ctl.Enabled Then
            If ctl.ControlType <> acLabel Then
                On Error Resume Next
                If Not ctl.Locked Then
                    If ctl.TabIndex > idx And ctl.TabIndex < nextIdx Then
                        nextIdx = ctl.TabIndex
                        nextName = ctl.name
                    End If
                    If ctl.TabIndex < firstIdx Then
                        firstIdx = ctl.TabIndex
                        firstName = ctl.name
                    End If
                End If
                On Error GoTo 0
            End If
        End If
    Next ctl

    If Len(nextName) = 0 Then nextName = firstName
    NextEditableControlName = nextName
End Function

' Mueve el foco al próximo control editable (según TabIndex)
Public Sub MoveToNextEditable(frm As Form, fromCtl As Control)
    Dim tgt As String
    tgt = NextEditableControlName(frm, fromCtl)
    If Len(tgt) > 0 Then
        DoCmd.GoToControl tgt
    End If
End Sub

Public Sub MoveToNextRowSameColumn(frm As Form, ctlName As String, Optional wrap As Boolean = False)
    On Error GoTo EH

    If frm.Dirty Then frm.Dirty = False

    DoCmd.GoToRecord , , acNext
    frm.Controls(ctlName).SetFocus
    Exit Sub

EH:
    Err.Clear
    If wrap Then
        On Error Resume Next
        DoCmd.GoToRecord , , acFirst
        frm.Controls(ctlName).SetFocus
        Exit Sub
    End If

    On Error Resume Next
    frm.Controls(ctlName).SetFocus
End Sub

'=========================================================
Public Sub RecalcTotalAplicado_SF(ByVal idDet As Long)
    On Error GoTo ErrH

    If idDet <= 0 Then Exit Sub

    Dim db As DAO.Database
    Set db = CurrentDb

    ' Asegura tabla TEMP local (Access) para el cálculo
    EnsureTmpTotalesAplicado

    ' Limpio e inserto el total calculado (solo 1 fila)
    db.Execute "DELETE FROM TMP_TotalesAplicado;", dbFailOnError

    db.Execute _
        "INSERT INTO TMP_TotalesAplicado (IdDetalleOrden, TotalAplicado) " & _
        "SELECT " & idDet & " AS IdDetalleOrden, Nz(Sum(CantidadAsignada),0) AS TotalAplicado " & _
        "FROM Ordenes_Detalles_Distrib " & _
        "WHERE IdDetalleOrden=" & idDet & ";", _
        dbFailOnError

    ' UPDATE con JOIN a tabla local => actualizable (aunque Ordenes_Detalles sea ODBC)
    db.Execute _
        "UPDATE Ordenes_Detalles AS D " & _
        "INNER JOIN TMP_TotalesAplicado AS T ON D.IdDetalleOrden = T.IdDetalleOrden " & _
        "SET D.[Total Aplicado] = Nz(T.TotalAplicado,0) " & _
        "WHERE D.IdDetalleOrden=" & idDet & ";", _
        dbFailOnError

    Exit Sub

ErrH:
    MsgBox "RecalcTotalAplicado_SF: " & Err.Number & " - " & Err.Description, vbCritical
End Sub

Public Sub Distrib_Proporcional_PorDetalle( _
        ByVal IdDetalleOrden As Long, _
        ByVal totalReal As Double, _
        Optional ByVal PasoRedondeo As Double = 0)

    On Error GoTo ErrH

    If IdDetalleOrden <= 0 Then Exit Sub

    Dim db As DAO.Database
    Dim qdf As DAO.QueryDef
    Dim sql As String
    Dim cn As String

    Set db = CurrentDb

    cn = CurrentDb.TableDefs("Ordenes_Detalles_Distrib").Connect

    sql = "EXEC dbo.sp_Distrib_Proporcional_PorDetalle " & _
          "@IdDetalleOrden = " & IdDetalleOrden & ", " & _
          "@TotalReal = " & SqlNum(totalReal) & ", " & _
          "@PasoRedondeo = " & SqlNum(PasoRedondeo) & ";"

    qdf.Connect = cn
    qdf.sql = sql
    qdf.ReturnsRecords = False

    Set db = Nothing
    Exit Sub

ErrH:
    MsgBox "Distrib_Proporcional_PorDetalle - Error " & Err.Number & ": " & Err.Description, vbExclamation
    Resume Salir

' Devuelve el nombre EXACTO del campo clave en Formulados (Nº REG / N° REG / NO REG).
Private Function FU_KeyFieldName() As String
    Dim tdf As DAO.TableDef, fld As DAO.Field
    Dim nm As String, canon As String
    On Error GoTo fin

    Set tdf = CurrentDb.TableDefs("Formulados")
    For Each fld In tdf.Fields
        nm = fld.name
        ' Normalizo para comparar: quito º y °, y espacios dobles
        canon = Replace(Replace(UCase$(nm), "º", ""), "°", "")
        canon = Replace(canon, "  ", " ")
        If InStr(1, canon, "N REG", vbTextCompare) > 0 _
           Or InStr(1, canon, "NO REG", vbTextCompare) > 0 _
           Or InStr(1, canon, "N° REG", vbTextCompare) > 0 Then
            FU_KeyFieldName = "[" & nm & "]"
            Exit Function
        End If
    Next
fin:
End Function


' Devuelve la UNIDAD para un IdFormulado (OD.IdFormulado), con fallback "KGS".
Public Function GetUnidad(ByVal idForm As Variant) As String
    Dim v As Variant, keyFld As String
    GetUnidad = "KGS"                       ' default para Semillas/Fertilizantes/no hallado

    If IsNull(idForm) Or Len(Trim$(Nz(idForm, ""))) = 0 Then Exit Function

    keyFld = FU_KeyFieldName()
    If Len(keyFld) = 0 Then keyFld = "[Nº REG]"  ' por si no lo detecta, usa el habitual

    ' 1) Intento NUMÉRICO (sin comillas). Protegido ante 3464.
    On Error Resume Next
    If IsNumeric(idForm) Then
        v = DLookup("UNIDAD", "Formulados", keyFld & "=" & CLng(idForm))
        If Err.Number <> 0 Then
            Err.Clear
            v = Null
        End If
    End If
    On Error GoTo 0

    ' 2) Si no encontró, intento TEXTO (con comillas y escapando).
    If IsNull(v) Or Len(Trim$(v & "")) = 0 Then
        On Error Resume Next
        v = DLookup("UNIDAD", "Formulados", _
                    keyFld & "='" & Replace(CStr(idForm), "'", "''") & "'")
        If Err.Number <> 0 Then
            Err.Clear
            v = Null
        End If
        On Error GoTo 0
    End If

    If Not IsNull(v) Then
        v = UCase$(Trim$(v & ""))
        If Len(v) > 0 Then GetUnidad = v
    End If
End Function

' ================== UNIDADES - MOTOR UNIFICADO ==================
' Usa:
'   - Unidad_Rules(Patron, Destino) para sinónimos (kg, kgs, kilo -> KGS)
'   - UnidadesMedida(Unidad, Nombre, Tipo) para códigos canónicos

' Helper local de comillas (no choca con SqlTxt)
Private Function QUM(ByVal s As String) As String
    QUM = "'" & Replace(Nz(s, ""), "'", "''") & "'"
End Function

Public Function Unidad_ResolverCanon(ByVal uRaw As Variant) As String
    On Error GoTo EH
    
    Dim t As String
    Dim sql As String
    Dim rs As DAO.Recordset
    
    t = UCase$(Trim$(Nz(uRaw, "")))
    If t = "" Or t = "-" Or t = "–" Then Exit Function
    
    ' 1) Si ya existe como código directo en UnidadesMedida, usarlo
    If DCount("*", "UnidadesMedida", "Unidad=" & QUM(t)) > 0 Then
        Unidad_ResolverCanon = t
        Exit Function
    End If
    
    ' 2) Buscar regla en Unidad_Rules:  't' LIKE Patrón
    sql = "SELECT TOP 1 Destino " & _
          "FROM Unidad_Rules " & _
          "WHERE " & QUM(t) & " LIKE Patron " & _
          "ORDER BY Len(Patron) DESC, Patron;"
    
    ' FIX SQL Server IDENTITY: dbSeeChanges
    Set rs = CurrentDb.OpenRecordset(sql, dbOpenSnapshot, dbSeeChanges)
    If Not rs.EOF Then
        Unidad_ResolverCanon = Nz(rs!Destino, "")
        rs.Close
        Exit Function
    End If
    rs.Close
    
    ' 3) Fallback
    Unidad_ResolverCanon = t
    Exit Function
    
EH:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Unidad_ResolverCanon = ""
End Function
' Inserta una unidad canónica en UnidadesMedida si no existe
' uniCodigo: código corto (KGS, LTS, TON, BDN, etc.)
' uniNombre: descripción legible (puede ser lo que escribió el usuario)
' uniTipo  : "Masa", "Volumen", "Conteo", etc.
Public Sub Unidad_AgregarSiNoExiste(ByVal uniCodigo As String, _
                                    ByVal uniNombre As String, _
                                    Optional ByVal uniTipo As String = "")
    Dim existe As Long
    Dim sql As String
    
    uniCodigo = UCase$(Trim$(Nz(uniCodigo, "")))
    If uniCodigo = "" Then Exit Sub
    
    existe = DCount("*", "UnidadesMedida", "Unidad=" & QUM(uniCodigo))
    If existe = 0 Then
        sql = "INSERT INTO UnidadesMedida (Unidad, Nombre, Tipo) VALUES (" & _
              QUM(uniCodigo) & ", " & _
              QUM(IIf(uniNombre = "", uniCodigo, uniNombre)) & ", " & _
              IIf(uniTipo = "", "Null", QUM(uniTipo)) & ");"
        CurrentDb.Execute sql, dbFailOnError
    End If
End Sub

' === Compatibilidad histórica: wrapper viejo que ahora usa Unidad_ResolverCanon ===
Public Function MapUnidadCompat(ByVal uRaw As Variant) As String
    Dim canon As String
    canon = Unidad_ResolverCanon(uRaw)
    
    If canon = "" Then
        MapUnidadCompat = "UN"   ' fallback definitivo
    Else
        MapUnidadCompat = canon
    End If
End Function

' === Devuelve unidad canónica partiendo del IdFormulado (tabla Formulados) ===
' Usa la UNIDAD del maestro y la pasa por MapUnidadCompat (-> Unidad_ResolverCanon)
Public Function GetUnidadCompatById(ByVal idForm As Variant) As String
    Dim keyFld As String, v As Variant
    GetUnidadCompatById = "KGS"                         ' fallback si no existe el formulado

    If IsNull(idForm) Or Len(Trim$(Nz(idForm, ""))) = 0 Then Exit Function

    keyFld = FU_KeyFieldName()
    If Len(keyFld) = 0 Then keyFld = "[Nº REG]"         ' por si no lo detecta

    On Error Resume Next
    ' Intento numérico
    If IsNumeric(idForm) Then
        v = DLookup("UNIDAD", "Formulados", keyFld & "=" & CLng(idForm))
    Else
        v = DLookup("UNIDAD", "Formulados", keyFld & "='" & Replace(CStr(idForm), "'", "''") & "'")
    End If
    If Err.Number <> 0 Then Err.Clear: v = Null
    On Error GoTo 0

    If Not IsNull(v) Then
        GetUnidadCompatById = MapUnidadCompat(v)
    End If
End Function

Public Function SqlListParams(ByVal sSQL As String) As String
    On Error GoTo Salida

    Dim qd As DAO.QueryDef
    Dim p As DAO.Parameter
    Dim s As String

    Set qd = CurrentDb.CreateQueryDef("", sSQL)

    For Each p In qd.Parameters
        If Len(s) > 0 Then s = s & ", "
        s = s & p.name
    Next

    SqlListParams = s

Salida:
    On Error Resume Next
    Set qd = Nothing
End Function

Public Function CheckQueryParamsOrStop(ByVal queryName As String, ByVal label As String) As Boolean
    On Error GoTo EH

    Dim qd As DAO.QueryDef
    Set qd = CurrentDb.QueryDefs(queryName)

    Dim p As DAO.Parameter
    Dim s As String
    Dim nm As String

    For Each p In qd.Parameters
        nm = Nz(p.name, "")

        If InStr(1, nm, "TempVars!", vbTextCompare) > 0 _
        Or InStr(1, nm, "Forms!", vbTextCompare) > 0 _
        Or InStr(1, nm, "Reports!", vbTextCompare) > 0 Then
            ' ignorar
        Else
            If Len(s) > 0 Then s = s & ", "
            s = s & nm
        End If
    Next

    Debug.Print "----------------------------"
    Debug.Print label & " (QueryDef)"
    Debug.Print "QUERY: " & queryName
    Debug.Print "PARAMS (no ignorados): " & s

    If Len(s) > 0 Then
        MsgBox "Consulta guardada con parámetros NO resueltos (" & label & "): " & s, vbExclamation
        CheckQueryParamsOrStop = False
        Exit Function
    End If

    CheckQueryParamsOrStop = True
    Exit Function

EH:
    MsgBox "Error leyendo parámetros de QueryDef (" & queryName & "): " & Err.Number & " - " & Err.Description, vbExclamation
    CheckQueryParamsOrStop = False
End Function

Public Sub ListarReferenciasEnObjetos(ByVal tableName As String)
    Dim ao As AccessObject
    Dim obj As Object
    Dim prop As Property
    Dim s As String

    Debug.Print "=============================="
    Debug.Print "REFERENCIAS EN FORMS/REPORTS: " & tableName
    Debug.Print "=============================="

    ' Forms
    For Each ao In CurrentProject.AllForms
        DoCmd.OpenForm ao.name, acDesign, , , , acHidden
        Set obj = Forms(ao.name)

        s = Nz(obj.RecordSource, "")
        If InStr(1, s, tableName, vbTextCompare) > 0 Then
            Debug.Print "FORM RecordSource: "; ao.name
        End If

        ' Controles (combos/listas)
        Dim ctl As Control
        For Each ctl In obj.Controls
            If ctl.ControlType = acComboBox Or ctl.ControlType = acListBox Then
                s = Nz(ctl.RowSource, "")
                If InStr(1, s, tableName, vbTextCompare) > 0 Then
                    Debug.Print "FORM RowSource: "; ao.name; " -> "; ctl.name
                End If
            End If
        Next ctl

        DoCmd.Close acForm, ao.name, acSaveNo
    Next ao

    ' Reports (solo RecordSource; controles es más pesado)
    For Each ao In CurrentProject.AllReports
        DoCmd.OpenReport ao.name, acViewDesign, , , acHidden
        Set obj = Reports(ao.name)

        s = Nz(obj.RecordSource, "")
        If InStr(1, s, tableName, vbTextCompare) > 0 Then
            Debug.Print "REPORT RecordSource: "; ao.name
        End If

        DoCmd.Close acReport, ao.name, acSaveNo
    Next ao
End Sub

Public Sub BuscarContenedoresDeFormulario(ByVal formHijo As String)
    Dim ao As AccessObject
    Dim frm As Form
    Dim ctl As Control
    Dim hit As Long

    For Each ao In CurrentProject.AllForms
        DoCmd.OpenForm ao.name, acDesign, , , acFormReadOnly, acHidden
        Set frm = Forms(ao.name)

        For Each ctl In frm.Controls
            If ctl.ControlType = acSubform Then
                If InStr(1, Nz(ctl.SourceObject, ""), "Form." & formHijo, vbTextCompare) > 0 _
                Or InStr(1, Nz(ctl.SourceObject, ""), "Formulario." & formHijo, vbTextCompare) > 0 _
                Or InStr(1, Nz(ctl.SourceObject, ""), formHijo, vbTextCompare) > 0 Then
                    Debug.Print "HIT | Padre:", ao.name, "| Control:", ctl.name, "| SourceObject:", ctl.SourceObject
                    hit = hit + 1
                End If
            End If
        Next ctl

        DoCmd.Close acForm, ao.name, acSaveNo
    Next ao

    Debug.Print "TOTAL HITS:", hit
End Sub
Public Sub AuditarReferenciasEnFormulario(ByVal formName As String, ByVal needle As String)
    Dim frm As Form
    Dim ctl As Control
    Dim hit As Long

    DoCmd.OpenForm formName, acDesign, , , acFormReadOnly, acHidden
    Set frm = Forms(formName)

    If InStr(1, Nz(frm.RecordSource, ""), needle, vbTextCompare) > 0 Then
        Debug.Print "FORM RecordSource HIT:", formName
        Debug.Print frm.RecordSource
        hit = hit + 1
    End If
    If InStr(1, Nz(frm.Filter, ""), needle, vbTextCompare) > 0 Then
        Debug.Print "FORM Filter HIT:", formName, "|", frm.Filter
        hit = hit + 1
    End If
    If InStr(1, Nz(frm.OrderBy, ""), needle, vbTextCompare) > 0 Then
        Debug.Print "FORM OrderBy HIT:", formName, "|", frm.OrderBy
        hit = hit + 1
    End If

    For Each ctl In frm.Controls
        On Error Resume Next
        If InStr(1, Nz(ctl.ControlSource, ""), needle, vbTextCompare) > 0 Then
            Debug.Print "CTL ControlSource HIT:", ctl.name, "|", ctl.ControlSource
            hit = hit + 1
        End If
        If InStr(1, Nz(ctl.RowSource, ""), needle, vbTextCompare) > 0 Then
            Debug.Print "CTL RowSource HIT:", ctl.name
            Debug.Print ctl.RowSource
            hit = hit + 1
        End If
        On Error GoTo 0
    Next ctl

    DoCmd.Close acForm, formName, acSaveNo
    Debug.Print "TOTAL HITS:", hit
End Sub
Public Function AgregarInsumoOrdenSQL( _
    ByVal IdOrden As Long, _
    ByVal IdFormulado As Long, _
    ByVal UnidadElegida As String) As Long

    On Error GoTo ErrH

    Dim db As DAO.Database
    Dim qdf As DAO.QueryDef
    Dim rs As DAO.Recordset
    Dim sql As String
    Dim cn As String
    Dim u As String

    AgregarInsumoOrdenSQL = 0

    If IdOrden <= 0 Then Exit Function
    If IdFormulado <= 0 Then Exit Function

    u = Replace(UCase$(Trim$(Nz(UnidadElegida, ""))), "'", "''")
    If Len(u) = 0 Then u = "UN"

    Set db = CurrentDb
    cn = CurrentDb.TableDefs("Ordenes_Detalles").Connect

    sql = "EXEC dbo.sp_Orden_AgregarInsumo " & _
          "@IdOrden = " & IdOrden & ", " & _
          "@IdFormulado = " & IdFormulado & ", " & _
          "@Unidad = '" & u & "';"

    Set qdf = db.CreateQueryDef("")
    qdf.Connect = cn
    qdf.sql = sql
    qdf.ReturnsRecords = True

    Set rs = qdf.OpenRecordset(dbOpenSnapshot)

    If Not rs.EOF Then
        AgregarInsumoOrdenSQL = Nz(rs!IdDetalleOrden, 0)
    End If

Salir:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Set qdf = Nothing
    Set db = Nothing
    Exit Function

ErrH:
    MsgBox "AgregarInsumoOrdenSQL: " & Err.Number & " - " & Err.Description, vbExclamation
    Resume Salir
End Function
